## 第一篇：集装箱机器人

### 1.1 产品定位

集装箱机器人是用于**集装箱装卸作业**的专用自动化设备，适用于20ft/40ft标准集装箱的货物吞吐，具备大跨度作业、高负载能力、高精度定位等特点。

@dataclass
class Pose6D:
    """6D位姿"""
    position: np.ndarray        # 位置 [x, y, z]
    orientation: np.ndarray     # 四元数 [w, x, y, z] 或 旋转矩阵
    
    @classmethod
    def from_matrix(cls, matrix: np.ndarray) -> Pose6D:
        """从4x4变换矩阵创建"""
        assert matrix.shape == (4, 4), "需要4x4变换矩阵"
        return cls(
            position=matrix[:3, 3],
            orientation=Rotation.from_matrix(matrix[:3, :3]).as_quat()  # xyzw
        )
    
    @classmethod
    def from_rpy(cls, position: List[float], rpy: List[float]) -> Pose6D:
        """从位置+RPY创建"""
        r = Rotation.from_euler('xyz', rpy)
        return cls(
            position=np.array(position),
            orientation=r.as_quat()
        )
    
    def to_matrix(self) -> np.ndarray:
        """转换为4x4变换矩阵"""
        r = Rotation.from_quat(self.orientation)  # xyzw
        matrix = np.eye(4)
        matrix[:3, :3] = r.as_matrix()
        matrix[:3, 3] = self.position
        return matrix
    
    def distance_to(self, other: Pose6D) -> float:
        """计算与另一个位姿的距离"""
        pos_dist = np.linalg.norm(self.position - other.position)
        r = Rotation.from_quat(self.orientation)
        other_r = Rotation.from_quat(other.orientation)
        rot_dist = np.abs(np.arccos(np.clip(2 * np.sum(r.as_quat() * other_r.as_quat())**2 - 1, -1, 1)))
        return pos_dist + 0.1 * rot_dist

        return len(self.joint_names)

    def num_points(self) -> int:
        return len(self.points)
    
    def get_joint_position(self, joint_index: int) -> np.ndarray:
        return np.array([p.positions[joint_index] for p in self.points])
    
    def get_duration(self) -> float:
        if not self.points:
            return 0.0
        return self.points[-1].time_from_start
    
    def sample(self, time: float) -> Tuple[np.ndarray, np.ndarray, np.ndarray]:
        """在指定时间采样"""
        if not self.points:
            raise ValueError("轨迹为空")
        if time <= self.points[0].time_from_start:
            return self.points[0].positions, self.points[0].velocities, self.points[0].accelerations
        if time >= self.points[-1].time_from_start:
            return self.points[-1].positions, self.points[-1].velocities, self.points[-1].accelerations
        
        for i in range(len(self.points) - 1):
            if self.points[i].time_from_start <= time <= self.points[i+1].time_from_start:
                alpha = (time - self.points[i].time_from_start) / \
                       (self.points[i+1].time_from_start - self.points[i].time_from_start)
                pos = self.points[i].positions + alpha * (self.points[i+1].positions - self.points[i].positions)
                vel = self.points[i].velocities + alpha * (self.points[i+1].velocities - self.points[i].velocities)
                acc = self.points[i].accelerations + alpha * (self.points[i+1].accelerations - self.points[i].accelerations)
                return pos, vel, acc
        return self.points[-1].positions, self.points[-1].velocities, self.points[-1].accelerations

class PlanningPerformanceMonitor:
    """性能监控器"""
    
    def __init__(self, window_size: int = 100):
        self._planning_times: List[float] = []
        self._trajectory_lengths: List[float] = []
        self._smoothing_scores: List[float] = []
        self._window_size = window_size
        self._start_time: Optional[float] = None
    
    def start_timer(self):
        """开始计时"""
        self._start_time = time.perf_counter()
    
    def stop_timer(self) -> float:
        """停止计时并记录"""
        if self._start_time is None:
            return 0.0
        elapsed = time.perf_counter() - self._start_time
        self._planning_times.append(elapsed)
        if len(self._planning_times) > self._window_size:
            self._planning_times.pop(0)
        self._start_time = None
        return elapsed
    
    def record_trajectory_metrics(self, trajectory: Trajectory):
        """记录轨迹指标"""
        if not trajectory.points:
            return
        
        # 计算轨迹长度
        total_length = 0.0
        for i in range(len(trajectory.points) - 1):
            diff = trajectory.points[i+1].positions - trajectory.points[i].positions
            total_length += np.linalg.norm(diff)
        self._trajectory_lengths.append(total_length)
        
        # 计算平滑度（基于加速度）
        if len(trajectory.points) >= 3:
            accelerations = [p.accelerations for p in trajectory.points if len(p.accelerations) > 0]
            if accelerations:
                avg_acc = np.mean([np.linalg.norm(a) for a in accelerations])
                self._smoothing_scores.append(1.0 / (1.0 + avg_acc))
        
        # 清理历史
        if len(self._trajectory_lengths) > self._window_size:
            self._trajectory_lengths.pop(0)
        if len(self._smoothing_scores) > self._window_size:
            self._smoothing_scores.pop(0)
    
    @property
    def avg_planning_time(self) -> float:
        return np.mean(self._planning_times) if self._planning_times else 0.0
    
    @property
    def p95_planning_time(self) -> float:
        return np.percentile(self._planning_times, 95) if self._planning_times else 0.0
    
    @property
    def avg_trajectory_length(self) -> float:
        return np.mean(self._trajectory_lengths) if self._trajectory_lengths else 0.0
    
    @property
    def avg_smoothing_score(self) -> float:
        return np.mean(self._smoothing_scores) if self._smoothing_scores else 0.0
    
    def get_report(self) -> Dict[str, float]:
        """获取性能报告"""

            'avg_planning_time_ms': self.avg_planning_time * 1000,
            'p95_planning_time_ms': self.p95_planning_time * 1000,
            'avg_trajectory_length': self.avg_trajectory_length,
            'avg_smoothing_score': self.avg_smoothing_score
        }

# src/motion_planner/global_planner/planner.py
"""
全局轨迹规划器
提供RRT、RRT*、PRM等多种采样规划算法
"""

from __future__ import annotations
from dataclasses import dataclass
from typing import List, Tuple, Optional, Dict, Callable
import numpy as np
from queue import PriorityQueue
import random
import math
from collections import defaultdict

from ..common.foundation import (
    JointState, Pose6D, JointLimits, Trajectory, TrajectoryPoint,
    KinematicsBase, CollisionWorld, RobotType

@dataclass
class PlanningResult:
    """规划结果"""
    success: bool
    trajectory: Optional[Trajectory]
    planning_time: float
    nodes_visited: int
    message: str = ""


class TreeNode:
    """搜索树节点"""
    __slots__ = ['state', 'parent', 'cost', 'action']
    
    def __init__(self, state: np.ndarray, parent: Optional['TreeNode'] = None,
                 cost: float = 0.0, action: Optional[np.ndarray] = None):
        self.state = state
        self.parent = parent
        self.cost = cost
        self.action = action
    
    def get_path(self) -> List[np.ndarray]:
        """获取从根到当前节点的路径"""
        path = []
        node = self
        while node is not None:
            path.append(node.state)
            node = node.parent
        return list(reversed(path))


class GlobalPlannerBase:
    """全局规划器基类"""
    
    def __init__(self, 
                 joint_limits: JointLimits,
                 collision_world: CollisionWorld,
                 kinematics: KinematicsBase,
                 max_iterations: int = 10000,
                 step_size: float = 0.1,
                 goal_bias: float = 0.05):
        self.joint_limits = joint_limits
        self.collision_world = collision_world
        self.kinematics = kinematics
        self.max_iterations = max_iterations
        self.step_size = step_size
        self.goal_bias = goal_bias
    
    @property
    def num_joints(self) -> int:
        return len(self.joint_limits.positions_lower)
    
    def sample_configuration(self, goal: Optional[np.ndarray] = None) -> np.ndarray:
        """采样关节配置"""
        if goal is not None and random.random() < self.goal_bias:
            return goal.copy()
        
        return np.random.uniform(
            self.joint_limits.positions_lower,
            self.joint_limits.positions_upper
        )
    
    def distance(self, q1: np.ndarray, q2: np.ndarray) -> float:
        """计算两个配置之间的距离"""
        return np.linalg.norm(q1 - q2)
    
    def check_limits(self, q: np.ndarray) -> bool:
        """检查关节限位"""
        return np.all(q >= self.joint_limits.positions_lower) and \
               np.all(q <= self.joint_limits.positions_upper)
    
    def check_collision(self, q: np.ndarray) -> bool:
        """检查碰撞"""
        is_collision, _ = self.collision_world.check_collision(
            q, self.kinematics.forward
        )
        return is_collision
    
    @abstractmethod
    def plan(self, start: np.ndarray, goal: np.ndarray) -> PlanningResult:
        """执行规划"""
        pass


class RRTConnect(GlobalPlannerBase):
    """
    RRT-Connect 规划器
    适用于快速路径搜索，适合散货机器人场景（速度优先）
    """
    
    def __init__(self, *args, extend_steps: int = 10, **kwargs):
        super().__init__(*args, **kwargs)
        self.extend_steps = extend_steps
    
    def _steer(self, from_node: TreeNode, to_point: np.ndarray) -> Tuple[TreeNode, bool]:
        """
        从from_node向to_point方向扩展一步
        
        Returns:
            (new_node, reached_goal)
        """
        direction = to_point - from_node.state
        dist = np.linalg.norm(direction)
        
        if dist < 1e-6:
            return from_node, False
        
        direction = direction / dist
        step = min(self.step_size, dist)
        new_state = from_node.state + direction * step
        
        if not self.check_limits(new_state):
            return from_node, False
        
        if self.check_collision(new_state):
            return from_node, False
        
        new_node = TreeNode(
            state=new_state,
            parent=from_node,
            cost=from_node.cost + step,
            action=direction * step
        )
        
        reached = self.distance(new_state, to_point) < self.step_size
        return new_node, reached
    
    def _build_tree(self, start: np.ndarray, goal: np.ndarray, 
                   goal_sample_rate: float) -> Tuple[Optional[TreeNode], List[TreeNode]]:
        """构建单棵树"""
        if self.check_collision(start) or self.check_collision(goal):
            return None, []
        
        root = TreeNode(state=start, cost=0.0)
        nodes = [root]
        
        for _ in range(self.max_iterations):
            # 采样
            if random.random() < goal_sample_rate:
                sample = goal.copy()
            else:
                sample = self.sample_configuration(goal)
            
            # 找最近节点
            nearest = min(nodes, key=lambda n: self.distance(n.state, sample))
            
            # 扩展
            new_node, reached = self._steer(nearest, sample)
            if new_node is not nearest:
                nodes.append(new_node)
                if reached or self.distance(new_node.state, sample) < self.step_size:
                    return new_node, nodes
        
        return None, nodes
    
    def plan(self, start: np.ndarray, goal: np.ndarray) -> PlanningResult:
        """执行RRT-Connect规划"""
        import time
        start_time = time.perf_counter()
        
        # 构建两棵树
        tree_a_root = TreeNode(state=start)
        tree_b_root = TreeNode(state=goal)
        tree_a = [tree_a_root]
        tree_b = [tree_b_root]
        
        for iteration in range(self.max_iterations):
            # 交替扩展
            if iteration % 2 == 0:
                extend_tree, connect_tree = tree_a, tree_b
                extend_root, connect_root = tree_a_root, tree_b_root
            else:
                extend_tree, connect_tree = tree_b, tree_a
                extend_root, connect_root = tree_b_root, tree_a_root
            
            # 采样
            sample = self.sample_configuration(connect_root.state)
            
            # 找最近节点
            nearest = min(extend_tree, key=lambda n: self.distance(n.state, sample))
            
            # 扩展
            new_node, _ = self._steer(nearest, sample)
            if new_node is not nearest:
                extend_tree.append(new_node)
                
                # 尝试连接
                connect_nearest = min(connect_tree, key=lambda n: self.distance(n.state, new_node.state))
                
                # 逐步连接
                current = new_node
                for _ in range(self.extend_steps):
                    direction = connect_nearest.state - current.state
                    dist = np.linalg.norm(direction)
                    if dist < self.step_size:
                        break
                    direction = direction / dist
                    next_state = current.state + direction * self.step_size
                    
                    if not self.check_limits(next_state) or self.check_collision(next_state):
                        break
                    
                    next_node = TreeNode(
                        state=next_state,
                        parent=current,
                        cost=current.cost + self.step_size
                    )
                    current = next_node
                    extend_tree.append(current)
                
                # 检查是否连接成功
                if self.distance(current.state, connect_nearest.state) < self.step_size * 2:
                    # 构建完整路径
                    path = current.get_path() + connect_nearest.get_path()
                    
                    # 创建轨迹
                    trajectory = self._path_to_trajectory(path)
                    planning_time = time.perf_counter() - start_time
                    
                    return PlanningResult(
                        success=True,
                        trajectory=trajectory,
                        planning_time=planning_time,
                        nodes_visited=len(tree_a) + len(tree_b),
                        message="规划成功"
                    )
        
        planning_time = time.perf_counter() - start_time
        return PlanningResult(
            success=False,
            trajectory=None,
            planning_time=planning_time,
            nodes_visited=len(tree_a) + len(tree_b),
            message="达到最大迭代次数"
        )
    
    def _path_to_trajectory(self, path: List[np.ndarray]) -> Trajectory:
        """将路径转换为轨迹"""
        points = []
        cumulative_time = 0.0
        
        for i, state in enumerate(path):
            if i == 0:
                velocities = np.zeros(len(state))
                accelerations = np.zeros(len(state))
            else:
                # 简单速度估计
                dt = 0.1
                velocities = (state - path[i-1]) / dt
                accelerations = np.zeros(len(state))
            
            points.append(TrajectoryPoint(
                positions=state.copy(),
                velocities=velocities,
                accelerations=accelerations,
                time_from_start=cumulative_time
            ))
            cumulative_time += 0.1
        
        joint_names = [f"joint_{i}" for i in range(len(path[0]))]
        return Trajectory(joint_names=joint_names, points=points)


class RRTStar(GlobalPlannerBase):
    """
    RRT* 规划器
    渐进最优，适用于集装箱机器人场景（精度优先）
    """
    
    def __init__(self, *args, rewire_radius: float = 0.5, **kwargs):
        super().__init__(*args, **kwargs)
        self.rewire_radius = rewire_radius
    
    def _get_nearby_nodes(self, nodes: List[TreeNode], 
                         state: np.ndarray) -> List[Tuple[TreeNode, float]]:
        """获取附近节点"""
        nearby = []
        for node in nodes:
            dist = self.distance(node.state, state)
            if dist < self.rewire_radius:
                nearby.append((node, dist))
        return nearby
    
    def _steer(self, from_node: TreeNode, to_point: np.ndarray) -> Tuple[Optional[TreeNode], float]:
        """扩展节点"""
        direction = to_point - from_node.state
        dist = np.linalg.norm(direction)
        
        if dist < 1e-6:
            return None, 0.0
        
        direction = direction / dist
        step = min(self.step_size, dist)
        new_state = from_node.state + direction * step
        
        if not self.check_limits(new_state):
            return None, 0.0
        
        if self.check_collision(new_state):
            return None, 0.0
        
        new_node = TreeNode(
            state=new_state,
            parent=from_node,
            cost=from_node.cost + step,
            action=direction * step
        )
        return new_node, step
    
    def plan(self, start: np.ndarray, goal: np.ndarray) -> PlanningResult:
        """执行RRT*规划"""
        import time
        start_time = time.perf_counter()
        
        if self.check_collision(start) or self.check_collision(goal):
            return PlanningResult(
                success=False, trajectory=None,
                planning_time=time.perf_counter() - start_time,
                nodes_visited=0, message="起点或终点碰撞"
            )
        
        # 初始化
        goal_tolerance = 0.05
        best_goal_node: Optional[TreeNode] = None
        
        root = TreeNode(state=start, cost=0.0)
        nodes = [root]
        
        for iteration in range(self.max_iterations):
            # 采样
            if iteration > 100 and best_goal_node is not None:
                # 偏向目标区域的采样
                sample = self._sample_near_goal(goal, radius=0.5)
            else:
                sample = self.sample_configuration(goal)
            
            # 找最近节点
            nearest = min(nodes, key=lambda n: self.distance(n.state, sample))
            
            # 扩展
            new_node, _ = self._steer(nearest, sample)
            if new_node is None:
                continue
            
            # 重连线（找最小成本父节点）
            nearby = self._get_nearby_nodes(nodes, new_node.state)
            for nearby_node, dist in nearby:
                new_cost = nearby_node.cost + dist
                if new_cost < new_node.cost:
                    # 检查新路径是否无碰撞
                    if not self._check_segment_collision(nearby_node.state, new_node.state):
                        new_node.parent = nearby_node
                        new_node.cost = new_cost
            
            nodes.append(new_node)
            
            # 重布电线（优化附近节点）
            for nearby_node, dist in nearby:
                new_cost = new_node.cost + dist
                if new_cost < nearby_node.cost:
                    if not self._check_segment_collision(new_node.state, nearby_node.state):
                        nearby_node.parent = new_node
                        nearby_node.cost = new_cost
            
            # 检查是否到达目标
            if self.distance(new_node.state, goal) < goal_tolerance:
                if best_goal_node is None or new_node.cost < best_goal_node.cost:
                    best_goal_node = new_node
            
            # 定期优化已有路径
            if iteration % 100 == 0 and best_goal_node is not None:
                self._improve_path(best_goal_node, nodes)
        
        if best_goal_node is None:
            return PlanningResult(
                success=False, trajectory=None,
                planning_time=time.perf_counter() - start_time,
                nodes_visited=len(nodes), message="未找到可行路径"
            )
        
        # 构建轨迹
        trajectory = self._path_to_trajectory(best_goal_node.get_path())
        return PlanningResult(
            success=True, trajectory=trajectory,
            planning_time=time.perf_counter() - start_time,
            nodes_visited=len(nodes), message="规划成功"
        )
    
    def _sample_near_goal(self, goal: np.ndarray, radius: float) -> np.ndarray:
        """在目标附近采样"""
        sample = goal + np.random.uniform(-radius, radius, len(goal))
        sample = np.clip(sample, self.joint_limits.positions_lower, 
                        self.joint_limits.positions_upper)
        return sample
    
    def _check_segment_collision(self, q1: np.ndarray, q2: np.ndarray) -> bool:
        """检查线段碰撞"""
        steps = int(np.linalg.norm(q2 - q1) / (self.step_size / 2)) + 1
        for i in range(steps):
            alpha = i / max(steps - 1, 1)
            q = q1 + alpha * (q2 - q1)
            if self.check_collision(q):
                return True
        return False
    
    def _improve_path(self, goal_node: TreeNode, nodes: List[TreeNode]):
        """尝试优化路径"""
        # 简化实现：遍历附近节点尝试缩短路径
        path = goal_node.get_path()
        improved = True
        while improved:
            improved = False
            for i in range(len(path) - 2):
                for j in range(i + 2, len(path)):
                    if not self._check_segment_collision(path[i], path[j]):
                        path = path[:i+1] + path[j:]
                        improved = True
                        break
                if improved:
                    break
    
    def _path_to_trajectory(self, path: List[np.ndarray]) -> Trajectory:
        """将路径转换为轨迹"""
        points = []
        cumulative_time = 0.0
        
        for i, state in enumerate(path):
            velocities = np.zeros(len(state)) if i == 0 else \
                        (state - path[i-1]) / 0.1
            points.append(TrajectoryPoint(
                positions=state.copy(),
                velocities=velocities,
                accelerations=np.zeros(len(state)),
                time_from_start=cumulative_time
            ))
            cumulative_time += 0.1
        
        joint_names = [f"joint_{i}" for i in range(len(path[0]))]
        return Trajectory(joint_names=joint_names, points=points)


class PRM(GlobalPlannerBase):
    """
    PRM (Probabilistic Roadmap) 规划器
    适合多查询场景，预构建地图后快速查询
    """
    
    def __init__(self, *args, num_samples: int = 1000, k_nearest: int = 10, **kwargs):
        super().__init__(*args, **kwargs)
        self.num_samples = num_samples
        self.k_nearest = k_nearest
        self.roadmap: Dict[int, List[Tuple[int, float]]] = {}
        self.samples: List[np.ndarray] = []
    
    def _build_roadmap(self):
        """构建 roadmap"""
        print(f"Building PRM roadmap with {self.num_samples} samples...")
        
        # 采样
        for i in range(self.num_samples):
            sample = self.sample_configuration()
            if not self.check_collision(sample):
                self.samples.append(sample)
        
        print(f"Generated {len(self.samples)} collision-free samples")
        
        # 构建邻接关系
        for i, sample in enumerate(self.samples):
            distances = [(j, self.distance(sample, other)) 
                        for j, other in enumerate(self.samples) 
                        if i != j]
            distances.sort(key=lambda x: x[1])
            
            self.roadmap[i] = []
            for j, dist in distances[:self.k_nearest]:
                if not self._check_edge_collision(sample, self.samples[j]):
                    self.roadmap[i].append((j, dist))
    
    def _check_edge_collision(self, q1: np.ndarray, q2: np.ndarray) -> bool:
        """检查边碰撞"""
        steps = int(np.linalg.norm(q2 - q1) / (self.step_size / 2)) + 1
        for i in range(steps):
            alpha = i / max(steps - 1, 1)
            q = q1 + alpha * (q2 - q1)
            if self.check_collision(q):
                return True
        return False
    
    def _find_nearest_sample(self, state: np.ndarray) -> int:
        """找最近采样点"""
        return min(range(len(self.samples)), 
                  key=lambda i: self.distance(state, self.samples[i]))
    
    def plan(self, start: np.ndarray, goal: np.ndarray) -> PlanningResult:
        """执行PRM规划"""
        import time
        start_time = time.perf_counter()
        
        # 懒构建：按需构建
        if not self.samples:
            self._build_roadmap()
        
        # 连接起点和终点
        start_idx = len(self.samples)
        self.samples.append(start)
        goal_idx = len(self.samples)
        self.samples.append(goal)
        
        # A* 搜索
        open_set = PriorityQueue()
        open_set.put((0, start_idx))
        came_from = {start_idx: None}
        g_score = {start_idx: 0.0}
        
        while not open_set.empty():
            _, current = open_set.get()
            
            if current == goal_idx:
                # 重建路径
                path = self._reconstruct_path(came_from, start_idx, goal_idx)
                trajectory = self._path_to_trajectory(path)
                return PlanningResult(
                    success=True, trajectory=trajectory,
                    planning_time=time.perf_counter() - start_time,
                    nodes_visited=len(self.samples), message="规划成功"
                )
            
            # 遍历邻居
            neighbors = self.roadmap.get(current, [])
            if current == start_idx:
                # 起点连接到采样点
                nearest = self._find_nearest_sample(start)
                if not self._check_edge_collision(start, self.samples[nearest]):
                    neighbors.append((nearest, self.distance(start, self.samples[nearest])))
            if current == goal_idx:
                # 终点连接到采样点
                nearest = self._find_nearest_sample(goal)
                if not self._check_edge_collision(goal, self.samples[nearest]):
                    neighbors.append((nearest, self.distance(goal, self.samples[nearest])))
            
            for neighbor, dist in neighbors:
                tentative_g = g_score.get(current, float('inf')) + dist
                if tentative_g < g_score.get(neighbor, float('inf')):
                    came_from[neighbor] = current
                    g_score[neighbor] = tentative_g
                    f_score = tentative_g + self.distance(self.samples[neighbor], goal)
                    open_set.put((f_score, neighbor))
        
        return PlanningResult(
            success=False, trajectory=None,
            planning_time=time.perf_counter() - start_time,
            nodes_visited=len(self.samples), message="未找到路径"
        )
    
    def _reconstruct_path(self, came_from: Dict, start: int, goal: int) -> List[np.ndarray]:
        """重建路径"""
        path = []
        current = goal
        while current is not None:
            path.append(self.samples[current])
            current = came_from.get(current)
        return list(reversed(path))
    
    def _path_to_trajectory(self, path: List[np.ndarray]) -> Trajectory:
        """将路径转换为轨迹"""
        points = []
        cumulative_time = 0.0
        for i, state in enumerate(path):
            points.append(TrajectoryPoint(
                positions=state.copy(),
                velocities=np.zeros(len(state)),
                accelerations=np.zeros(len(state)),
                time_from_start=cumulative_time
            ))
            cumulative_time += 0.1
        joint_names = [f"joint_{i}" for i in range(len(path[0]))]
        return Trajectory(joint_names=joint_names, points=points)

# src/motion_planner/local_optimizer/optimizer.py
"""
局部轨迹优化器
提供CHOMP、STOMP等轨迹优化算法
"""

from __future__ import annotations
from dataclasses import dataclass
from typing import List, Callable, Optional, Tuple
import numpy as np
from abc import ABC, abstractmethod
import random

from ..common.foundation import (
    Trajectory, TrajectoryPoint, JointLimits, CollisionWorld, KinematicsBase
)


@dataclass
class OptimizationConfig:
    """优化配置"""
    max_iterations: int = 100
    learning_rate: float = 0.01
    smoothness_weight: float = 0.5
    collision_weight: float = 1.0
    smoothness_factor: float = 0.5
    

class TrajectoryOptimizerBase(ABC):
    """轨迹优化器基类"""
    
    def __init__(self, 
                 trajectory: Trajectory,
                 joint_limits: JointLimits,
                 collision_world: CollisionWorld,
                 kinematics: KinematicsBase,
                 config: OptimizationConfig):
        self.trajectory = trajectory
        self.joint_limits = joint_limits
        self.collision_world = collision_world
        self.kinematics = kinematics
        self.config = config
    
    @abstractmethod
    def optimize(self) -> Trajectory:
        """执行优化"""
        pass
    
    def _calculate_smoothness_cost(self, positions: np.ndarray) -> Tuple[float, np.ndarray]:
        """计算平滑度代价及其梯度"""
        n = len(positions)
        if n < 3:
            return 0.0, np.zeros_like(positions)
        
        # 使用加速度范数作为平滑度指标
        accelerations = positions[2:] - 2 * positions[1:-1] + positions[:-2]
        cost = 0.5 * np.sum(accelerations ** 2)
        
        # 梯度（简化）
        gradient = np.zeros_like(positions)
        gradient[1:-1] = -2 * accelerations + 2 * np.roll(accelerations, 1)[:-2]
        gradient[1:-1] = np.clip(gradient[1:-1], -1, 1)
        
        return cost, gradient * self.config.smoothness_weight
    
    def _calculate_collision_cost(self, positions: np.ndarray) -> Tuple[float, np.ndarray]:
        """计算碰撞代价"""
        total_cost = 0.0
        gradient = np.zeros_like(positions)
        
        collision_padding = 0.01  # 安全裕度
        
        for i, pos in enumerate(positions):
            is_collision, _ = self.collision_world.check_collision(
                pos, self.kinematics.forward
            )
            if is_collision:
                # 二次惩罚
                cost = collision_padding ** 2
                total_cost += cost
                gradient[i] = np.zeros(len(pos))  # 简化
        
        return total_cost * self.config.collision_weight, gradient
    
    def _enforce_limits(self, positions: np.ndarray) -> np.ndarray:
        """强制执行关节限位"""
        return np.clip(
            positions,
            self.joint_limits.positions_lower,
            self.joint_limits.positions_upper
        )


class CHOMPOptimizer(TrajectoryOptimizerBase):
    """
    CHOMP (Covariant Hamiltonian Optimization for Motion Planning)
    基于梯度下降的轨迹优化算法，适合集装箱场景（精度优先）
    """
    
    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        self.obstacle_cost_function: Optional[Callable] = None
    
    def _compute_obstacle_gradient(self, positions: np.ndarray) -> np.ndarray:
        """计算障碍物梯度"""
        gradient = np.zeros_like(positions)
        
        if self.obstacle_cost_function is None:
            return gradient
        
        for i in range(len(positions)):
            eps = 1e-5
            pos = positions[i]
            
            # 数值梯度
            grad = np.zeros(len(pos))
            for j in range(len(pos)):
                pos_plus = pos.copy()
                pos_plus[j] += eps
                pos_minus = pos.copy()
                pos_minus[j] -= eps
                
                cost_plus = self._single_point_collision_cost(pos_plus)
                cost_minus = self._single_point_collision_cost(pos_minus)
                grad[j] = (cost_plus - cost_minus) / (2 * eps)
            
            gradient[i] = grad
        
        return gradient
    
    def _single_point_collision_cost(self, pos: np.ndarray) -> float:
        """单点碰撞代价"""
        is_collision, _ = self.collision_world.check_collision(
            pos, self.kinematics.forward
        )
        return 100.0 if is_collision else 0.0
    
    def optimize(self) -> Trajectory:
        """执行CHOMP优化"""
        positions = np.array([p.positions for p in self.trajectory.points])
        velocities = np.array([p.velocities for p in self.trajectory.points])
        
        n_points = len(positions)
        n_joints = len(positions[0])
        
        # 初始化速度
        if all(len(v) == 0 for v in velocities):
            velocities = np.zeros_like(positions)
        
        dt = 0.1  # 时间步长
        
        for iteration in range(self.config.max_iterations):
            # 计算梯度
            smoothness_cost, smoothness_grad = self._calculate_smoothness_cost(positions)
            collision_cost, collision_grad = self._calculate_collision_cost(positions)
            obstacle_grad = self._compute_obstacle_gradient(positions)
            
            # 总梯度
            total_grad = (
                self.config.smoothness_weight * smoothness_grad +
                self.config.collision_weight * collision_grad +
                obstacle_grad
            )
            
            # 梯度下降更新
            eta = self.config.learning_rate / (1 + iteration * 0.01)
            velocities = velocities - eta * total_grad
            
            # 更新位置
            positions = positions + dt * velocities
            
            # 强制限位
            positions = self._enforce_limits(positions)
            
            # 重新计算速度
            velocities = np.gradient(positions, axis=0, edge_order=2)
        
        # 构建优化后的轨迹
        points = []
        for i, pos in enumerate(positions):
            t = i * dt
            vel = velocities[i] if i < len(velocities) else np.zeros(n_joints)
            points.append(TrajectoryPoint(
                positions=pos,
                velocities=vel,
                accelerations=np.zeros(n_joints),
                time_from_start=t
            ))
        
        return Trajectory(
            joint_names=self.trajectory.joint_names,
            points=points
        )


class STOMPOptimizer(TrajectoryOptimizerBase):
    """
    STOMP (Stochastic Trajectory Optimization for Motion Planning)
    基于随机优化的轨迹优化算法，适合散货场景（速度优先）
    """
    
    def __init__(self, *args, 
                 num_trajectories: int = 10,
                 exploration_variance: float = 1.0,
                 **kwargs):
        super().__init__(*args, **kwargs)
        self.num_trajectories = num_trajectories
        self.exploration_variance = exploration_variance
    
    def _compute_noisy_trajectories(self, base_positions: np.ndarray) -> List[np.ndarray]:
        """生成噪声轨迹"""
        n_points, n_joints = base_positions.shape
        trajectories = []
        
        for _ in range(self.num_trajectories):
            # 添加高斯噪声
            noise = np.random.normal(0, self.exploration_variance, base_positions.shape)
            
            # 平滑噪声
            from scipy.ndimage import gaussian_filter1d
            noise_smoothed = gaussian_filter1d(noise, sigma=2, axis=0)
            
            noisy_trajectory = base_positions + noise_smoothed
            noisy_trajectory = self._enforce_limits(noisy_trajectory)
            trajectories.append(noisy_trajectory)
        
        return trajectories
    
    def _compute_trajectory_cost(self, positions: np.ndarray) -> float:
        """计算轨迹总代价"""
        # 平滑度代价
        smoothness_cost, _ = self._calculate_smoothness_cost(positions)
        
        # 碰撞代价
        collision_cost, _ = self._calculate_collision_cost(positions)
        
        # 限位代价
        limit_cost = 0.0
        near_limit = 0.1  # 接近限位的裕度
        
        for pos in positions:
            lower_diff = self.joint_limits.positions_lower - pos
            upper_diff = pos - self.joint_limits.positions_upper
            
            violation = np.maximum(0, -near_limit - lower_diff) + \
                       np.maximum(0, -near_limit - upper_diff)
            limit_cost += np.sum(violation ** 2)
        
        return smoothness_cost * self.config.smoothness_weight + \
               collision_cost * self.config.collision_weight + \
               limit_cost * 10.0
    
    def _update_trajectory(self, base_positions: np.ndarray, 
                          costs: np.ndarray,
                          noisy_trajectories: List[np.ndarray]) -> np.ndarray:
        """基于概率加权更新轨迹"""
        # softmax加权
        weights = np.exp(-costs)
        weights = weights / np.sum(weights)
        
        # 加权平均
        updated = np.zeros_like(base_positions)
        for weight, trajectory in zip(weights, noisy_trajectories):
            updated += weight * trajectory
        
        return updated
    
    def optimize(self) -> Trajectory:
        """执行STOMP优化"""
        base_positions = np.array([p.positions for p in self.trajectory.points])
        n_points, n_joints = base_positions.shape
        
        # 初始化速度
        velocities = np.array([
            p.velocities if len(p.velocities) > 0 else np.zeros(n_joints)
            for p in self.trajectory.points
        ])
        
        dt = 0.1
        best_cost = float('inf')
        best_positions = base_positions.copy()
        
        for iteration in range(self.config.max_iterations):
            # 生成噪声轨迹
            noisy_trajectories = self._compute_noisy_trajectories(base_positions)
            
            # 计算每条轨迹的代价
            costs = np.array([self._compute_trajectory_cost(traj) 
                            for traj in noisy_trajectories])
            
            # 更新基础轨迹
            base_positions = self._update_trajectory(base_positions, costs, noisy_trajectories)
            base_positions = self._enforce_limits(base_positions)
            
            # 记录最优
            current_cost = self._compute_trajectory_cost(base_positions)
            if current_cost < best_cost:
                best_cost = current_cost
                best_positions = base_positions.copy()
            
            # 衰减探索方差
            self.exploration_variance *= 0.95
        
        # 使用最优轨迹
        velocities = np.gradient(best_positions, axis=0, edge_order=2)
        
        # 构建优化后的轨迹
        points = []
        for i, pos in enumerate(best_positions):
            t = i * dt
            vel = velocities[i] if i < len(velocities) else np.zeros(n_joints)
            points.append(TrajectoryPoint(
                positions=pos,
                velocities=vel,
                accelerations=np.zeros(n_joints),
                time_from_start=t
            ))
        
        return Trajectory(
            joint_names=self.trajectory.joint_names,
            points=points
        )


class TrajectoryOptimizerFactory:
    """轨迹优化器工厂"""
    
    OPTIMIZER_MAP = {
        'container': CHOMPOptimizer,  # 集装箱：高精度平滑
        'bulk_cargo': STOMPOptimizer,  # 散货：快速优化

"""
轨迹插补与下发模块
提供S曲线插补、线性插补等算法，并负责下发至机器人控制器

class InterpolationMode(Enum):
    """插补模式"""
    LINEAR = "linear"                    # 线性插补（散货场景）
    S_CURVE = "s_curve"                  # S曲线插补（集装箱场景）


@dataclass
class InterpolationConfig:
    """插补配置"""
    control_frequency: int = 125        # 控制频率 (Hz) - 集装箱
    max_velocity_scale: float = 0.8       # 最大速度比例
    max_acceleration_scale: float = 0.6  # 最大加速度比例
    s_curve_jerk_limit: float = 100.0     # S曲线加加速度限制


class TrajectoryInterpolator:
    """
    轨迹插补器
    将优化后的轨迹进行时间参数化，生成控制指令
    """
    
    def __init__(self, 
                 joint_limits: JointLimits,
                 config: Optional[InterpolationConfig] = None):
        self.joint_limits = joint_limits
        self.config = config or InterpolationConfig()
        self._interpolation_mode = InterpolationMode.S_CURVE
    
    def set_mode(self, mode: InterpolationMode):
        """设置插补模式"""
        self._interpolation_mode = mode
        if mode == InterpolationMode.LINEAR:
            self.config.control_frequency = 250  # 散货场景提高频率
        else:
            self.config.control_frequency = 125  # 集装箱场景
    
    def interpolate_linear(self, trajectory: Trajectory) -> List[JointState]:
        """
        线性插补（散货场景 - 高速连续运动）
        
        特点：计算简单，响应快，适合节拍要求高的场景
        """
        if not trajectory.points or len(trajectory.points) < 2:
            return []
        
        dt = 1.0 / self.config.control_frequency
        joint_states = []
        
        start_point = trajectory.points[0]
        end_point = trajectory.points[-1]
        total_duration = end_point.time_from_start
        
        num_steps = int(total_duration / dt)
        
        for i in range(num_steps + 1):
            t = i * dt
            alpha = min(t / total_duration, 1.0)
            
            # 线性插值
            positions = (1 - alpha) * start_point.positions + alpha * end_point.positions
            
            # 简化速度计算
            if i == 0:
                velocities = np.zeros(len(positions))
                accelerations = np.zeros(len(positions))
            else:
                velocities = (positions - joint_states[-1].positions) / dt
                accelerations = np.zeros(len(positions))
            
            # 应用速度/加速度限制
            velocities = self._limit_velocities(velocities, positions)
            
            joint_states.append(JointState(
                positions=positions,
                velocities=velocities,
                accelerations=accelerations,
                timestamp=t
            ))
        
        return joint_states
    
    def interpolate_s_curve(self, trajectory: Trajectory) -> List[JointState]:
        """
        S曲线插补（集装箱场景 - 平稳精确运动）
        
        特点：加减速平滑，冲击小，适合大负载高精度场景
        """
        if not trajectory.points or len(trajectory.points) < 2:
            return []
        
        # 从轨迹点提取路径
        waypoints = np.array([p.positions for p in trajectory.points])
        durations = np.array([p.time_from_start for p in trajectory.points])
        
        # 计算每个关节的速度/加速度限制
        v_max = self.joint_limits.velocities * self.config.max_velocity_scale
        a_max = self.joint_limits.accelerations * self.config.max_acceleration_scale
        
        # S曲线时间参数化
        s_curve_params = self._compute_s_curve_parameters(waypoints, v_max, a_max)
        
        dt = 1.0 / self.config.control_frequency
        joint_states = []
        t = 0.0
        
        while t <= s_curve_params['total_duration'] + 1e-6:
            s = self._evaluate_s_curve(t, s_curve_params)
            
            # 通过s值插值关节位置
            positions = self._interpolate_by_arc_length(waypoints, s)
            
            # 计算速度
            ds_dt = s_curve_params.get('velocity', 1.0)  # 简化
            velocities = np.zeros(len(positions))  # 实际应从s计算
            
            # 应用限制
            velocities = self._limit_velocities(velocities, positions)
            
            joint_states.append(JointState(
                positions=positions,
                velocities=velocities,
                accelerations=np.zeros(len(positions)),
                timestamp=t
            ))
            
            t += dt
        
        return joint_states
    
    def _compute_s_curve_parameters(self, 
                                    waypoints: np.ndarray,
                                    v_max: np.ndarray,
                                    a_max: np.ndarray) -> dict:
        """计算S曲线参数"""
        # 计算相邻点之间的距离
        segment_lengths = np.array([
            np.linalg.norm(waypoints[i+1] - waypoints[i])
            for i in range(len(waypoints) - 1)
        ])
        
        total_length = np.sum(segment_lengths)
        
        # 基于加速度限制计算加速段时间
        # t_a = v_max / a_max (简化)
        jerk_limit = self.config.s_curve_jerk_limit
        
        # 计算各段时间
        # 简化：假设匀加速、匀速、匀减速三段
        v_avg = total_length / 10.0  # 估计平均速度
        t_accel = v_avg / np.max(a_max)
        t_decel = v_avg / np.max(a_max)
        t_constant = max(0, (total_length - v_avg * (t_accel + t_decel)) / v_avg)
        
        return {
            'total_duration': t_accel + t_constant + t_decel,
            't_accel': t_accel,
            't_constant': t_constant,
            't_decel': t_decel,
            'v_max': v_avg,
            'a_max': np.max(a_max),
            'jerk_limit': jerk_limit
        }
    
    def _evaluate_s_curve(self, t: float, params: dict) -> float:
        """评估S曲线位置（归一化0-1）"""
        T = params['total_duration']
        t_a = params['t_accel']
        t_c = params['t_constant']
        t_d = params['t_decel']
        
        if t <= 0:
            return 0.0
        elif t <= t_a:
            # 加速段
            ratio = t / t_a
            return 0.5 * ratio * ratio
        elif t <= t_a + t_c:
            # 匀速段
            ratio = (t - t_a) / t_c
            return 0.5 + 0.5 * ratio
        elif t <= T:
            # 减速段
            ratio = (t - t_a - t_c) / t_d
            return 1.0 - 0.5 * (1.0 - ratio) * (1.0 - ratio)
        else:
            return 1.0
    
    def _interpolate_by_arc_length(self, waypoints: np.ndarray, s: float) -> np.ndarray:
        """基于弧长参数s插值位置"""
        n_segments = len(waypoints) - 1
        if n_segments == 0:
            return waypoints[0]
        
        # 计算累计弧长
        segment_lengths = np.array([
            np.linalg.norm(waypoints[i+1] - waypoints[i])
            for i in range(n_segments)
        ])
        cumulative_lengths = np.cumsum(segment_lengths)
        total_length = cumulative_lengths[-1]
        
        # 目标弧长
        target_length = s * total_length
        
        # 找到所在段
        segment_idx = np.searchsorted(cumulative_lengths, target_length)
        segment_idx = min(segment_idx, n_segments - 1)
        
        # 段内插值
        start_length = cumulative_lengths[segment_idx - 1] if segment_idx > 0 else 0
        local_s = (target_length - start_length) / segment_lengths[segment_idx]
        
        return (1 - local_s) * waypoints[segment_idx] + local_s * waypoints[segment_idx + 1]
    
    def _limit_velocities(self, velocities: np.ndarray, 
                         positions: np.ndarray) -> np.ndarray:
        """限制速度"""
        v_max = self.joint_limits.velocities * self.config.max_velocity_scale
        return np.clip(velocities, -v_max, v_max)
    
    def interpolate(self, trajectory: Trajectory) -> List[JointState]:
        """执行插补"""
        if self._interpolation_mode == InterpolationMode.LINEAR:
            return self.interpolate_linear(trajectory)
        else:
            return self.interpolate_s_curve(trajectory)

class TrajectoryExecutionMonitor:
    """轨迹执行监控器"""
    
    def __init__(self, controller: RobotControllerInterface):
        self.controller = controller
        self._execution_start_time: Optional[float] = None
        self._expected_duration: float = 0.0
        self._executed_points: int = 0
    
    def start_monitoring(self, joint_states: List[JointState], expected_duration: float):
        """开始监控"""
        self._execution_start_time = time.time()
        self._expected_duration = expected_duration
        self._executed_points = 0
    
    def update(self) -> dict:
        """更新监控状态"""
        elapsed = time.time() - self._execution_start_time if self._execution_start_time else 0
        progress = min(elapsed / self._expected_duration, 1.0) if self._expected_duration > 0 else 0
        
        current_state = self.controller.get_current_state()
        
        return {
            'elapsed_time': elapsed,
            'expected_duration': self._expected_duration,
            'progress': progress,
            'current_positions': current_state.positions,
            'current_velocities': current_state.velocities,
            'is_on_track': self._check_tracking_error(current_state)
        }
    
    def _check_tracking_error(self, current_state: JointState) -> bool:
        """检查跟踪误差"""
        if not self.controller._command_buffer:
            return True
        
        expected = self.controller._command_buffer[min(
            int(current_state.timestamp * 125),  # 假设125Hz
            len(self.controller._command_buffer) - 1
        )]
        
        error = np.linalg.norm(current_state.positions - expected.positions)
        return error < 0.01  # 1cm容差

# src/motion_planner/motion_planner.py
"""
运动规划器主模块
整合全局规划、局部优化、轨迹插补，提供统一接口
"""

from __future__ import annotations
from dataclasses import dataclass
from typing import Optional, List, Tuple
import numpy as np

from .common.foundation import (
    JointState, Pose6D, JointLimits, Trajectory, TrajectoryPoint,
    CollisionWorld, KinematicsBase, RobotType, PlanningPerformanceMonitor
)
from .global_planner.planner import (
    GlobalPlannerBase, RRTConnect, RRTStar, PRM, StrategySelector, PlanningResult
)
from .local_optimizer.optimizer import (
    TrajectoryOptimizerBase, CHOMPOptimizer, STOMPOptimizer,
    TrajectoryOptimizerFactory, OptimizationConfig
)
from .interpolation.interpolator import (
    TrajectoryInterpolator, RobotControllerInterface, 
    TrajectoryExecutionMonitor, InterpolationConfig, InterpolationMode
)


@dataclass
class MotionPlanRequest:
    """运动规划请求"""
    start_joints: np.ndarray
    goal_joints: np.ndarray
    robot_type: RobotType
    priority: str = "normal"  # "speed" / "precision" / "normal"


@dataclass
class MotionPlanResponse:
    """运动规划响应"""
    success: bool
    trajectory: Optional[Trajectory]
    joint_states: Optional[List[JointState]]
    planning_time: float
    message: str


class MotionPlanner:
    """
    运动规划器主类
    
    提供完整的运动规划流程：
    1. 全局轨迹规划（RRT/RRT*/PRM）
    2. 局部轨迹优化（CHOMP/STOMP）
    3. 轨迹插补与下发（S曲线/线性）
    """
    
    def __init__(self,
                 joint_limits: JointLimits,
                 collision_world: CollisionWorld,
                 kinematics: KinematicsBase,
                 robot_type: RobotType,
                 controller: Optional[RobotControllerInterface] = None):
        self.joint_limits = joint_limits
        self.collision_world = collision_world
        self.kinematics = kinematics
        self.robot_type = robot_type
        self.controller = controller
        
        # 初始化各层组件
        self._init_global_planner()
        self._init_local_optimizer()
        self._init_interpolator()
        
        # 性能监控
        self.performance_monitor = PlanningPerformanceMonitor()
    
    def _init_global_planner(self):
        """初始化全局规划器"""
        self.global_planner: GlobalPlannerBase = StrategySelector.select_planner(
            robot_type=self.robot_type,
            joint_limits=self.joint_limits,
            collision_world=self.collision_world,
            kinematics=self.kinematics,
            max_iterations=5000,
            step_size=0.1
        )
    
    def _init_local_optimizer(self):
        """初始化局部优化器"""
        self.local_optimizer: TrajectoryOptimizerBase = \
            TrajectoryOptimizerFactory.create_optimizer(
                robot_type=self.robot_type.value,
                trajectory=None,  # 稍后设置
                joint_limits=self.joint_limits,
                collision_world=self.collision_world,
                kinematics=self.kinematics,
                config=OptimizationConfig(
                    max_iterations=50,
                    smoothness_weight=0.5 if self.robot_type == RobotType.CONTAINER else 0.3,
                    collision_weight=2.0 if self.robot_type == RobotType.CONTAINER else 1.0
                )
            )
    
    def _init_interpolator(self):
        """初始化插补器"""
        config = InterpolationConfig(
            control_frequency=125 if self.robot_type == RobotType.CONTAINER else 250,
            max_velocity_scale=0.8 if self.robot_type == RobotType.CONTAINER else 1.0,
            max_acceleration_scale=0.6 if self.robot_type == RobotType.CONTAINER else 0.9
        )
        self.interpolator = TrajectoryInterpolator(
            joint_limits=self.joint_limits,
            config=config
        )
        self.interpolator.set_mode(
            InterpolationMode.S_CURVE if self.robot_type == RobotType.CONTAINER 
            else InterpolationMode.LINEAR
        )
    
    def plan(self, request: MotionPlanRequest) -> MotionPlanResponse:
        """
        执行完整运动规划
        
        流程：全局规划 -> 局部优化 -> 轨迹插补
        """
        import time
        total_start = time.perf_counter()
        
        # 1. 全局轨迹规划
        self.performance_monitor.start_timer()
        global_result: PlanningResult = self.global_planner.plan(
            start=request.start_joints,
            goal=request.goal_joints
        )
        global_time = self.performance_monitor.stop_timer()
        
        if not global_result.success:
            return MotionPlanResponse(
                success=False,
                trajectory=None,
                joint_states=None,
                planning_time=time.perf_counter() - total_start,
                message=f"全局规划失败: {global_result.message}"
            )
        
        # 2. 局部轨迹优化
        self.performance_monitor.start_timer()
        self.local_optimizer.trajectory = global_result.trajectory
        optimized_trajectory: Trajectory = self.local_optimizer.optimize()
        optimize_time = self.performance_monitor.stop_timer()
        
        # 记录轨迹指标
        self.performance_monitor.record_trajectory_metrics(optimized_trajectory)
        
        # 3. 轨迹插补
        joint_states = self.interpolator.interpolate(optimized_trajectory)
        
        total_time = time.perf_counter() - total_start
        
        return MotionPlanResponse(
            success=True,
            trajectory=optimized_trajectory,
            joint_states=joint_states,
            planning_time=total_time,
            message=f"规划成功 (全局:{global_time*1000:.1f}ms + 优化:{optimize_time*1000:.1f}ms)"
        )
    
    def plan_to_pose(self, target_pose: Pose6D, 
                     seed: Optional[np.ndarray] = None) -> MotionPlanResponse:
        """规划到目标位姿（自动IK）"""
        # 逆运动学求解
        if seed is None:
            seed = self.kinematics.forward(self.joint_limits.positions_lower).position
        
        try:
            goal_joints = self.kinematics.inverse(target_pose, seed)
        except Exception as e:
            return MotionPlanResponse(
                success=False,
                trajectory=None,
                joint_states=None,
                planning_time=0,
                message=f"IK求解失败: {str(e)}"
            )
        
        current_state = self.controller.get_current_state() if self.controller else None
        start_joints = current_state.positions if current_state else self.joint_limits.positions_lower
        
        return self.plan(MotionPlanRequest(
            start_joints=start_joints,
            goal_joints=goal_joints,
            robot_type=self.robot_type
        ))
    
    def execute(self, response: MotionPlanResponse, 
               monitor: bool = True) -> bool:
        """执行规划结果"""
        if not response.success or response.joint_states is None:
            print(f"Execution failed: {response.message}")
            return False
        
        if self.controller is None:
            print("No controller connected")
            return False
        
        # 连接控制器
        if not self.controller._connected:
            # 默认连接
            self.controller.connect("192.168.1.100")
        
        # 发送轨迹
        monitor_obj = None
        if monitor:
            monitor_obj = TrajectoryExecutionMonitor(self.controller)
            monitor_obj.start_monitoring(
                response.joint_states,
                response.trajectory.get_duration()
            )
        
        success = self.controller.send_trajectory(
            response.joint_states,
            blocking=True
        )
        
        if monitor and monitor_obj:
            final_status = monitor_obj.update()
            print(f"Execution complete: {final_status}")
        
        return success
    
    def get_performance_report(self) -> dict:
        """获取性能报告"""
        return self.performance_monitor.get_report()
```

        深度图处理

        depth_m[depth_m > self.config.depth_max] = 0
        depth_m = np.nan_to_num(depth_m, nan=0.0)
        
        # 双边滤波保边去噪
        if self.config.bilateral_sigma_depth > 0:
            depth_m = bilateral_filter(
                depth_m.astype(np.float64),
                sigma_depth=self.config.bilateral_sigma_depth,
                sigma_spatial=self.config.bilateral_sigma_space
            )
        
        # 小孔填充（可选）
        depth_m = self._fill_holes(depth_m)
        
        return depth_m.astype(np.float32)
    
    def _fill_holes(self, depth: np.ndarray, 
                    max_hole_size: int = 100) -> np.ndarray:
        """填充深度图中的小孔"""
        mask = (depth > 0).astype(np.uint8)
        contours, _ = cv2.findContours(mask, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
        
        filled = depth.copy()
        
        for contour in contours:
            area = cv2.contourArea(contour)
            if area < max_hole_size:
                # 使用最近邻填充
                mask_fill = np.zeros_like(mask)
                cv2.drawContours(mask_fill, [contour], -1, 1, -1)
                
                # 找最近有效像素值
                valid_mask = (mask == 1) & (mask_fill == 1)
                if np.any(valid_mask):
                    valid_depths = depth[valid_mask]
                    fill_value = np.median(valid_depths)
                    filled[mask_fill == 1] = fill_value
        
        return filled
    
    def extract_roi(self, depth: np.ndarray, 
                   roi_bounds: Tuple[int, int, int, int]) -> np.ndarray:
        """
        提取感兴趣区域
        
        Args:
            roi_bounds: (x_min, y_min, x_max, y_max)
        """
        x_min, y_min, x_max, y_max = roi_bounds
        return depth[y_min:y_max, x_min:x_max].copy()
    
    def create_masked_depth(self, depth: np.ndarray, 
                           mask: np.ndarray) -> np.ndarray:
        """创建带掩码的深度图"""
        masked = depth.copy()
        masked[mask == 0] = 0
        return masked


class PointCloudProcessor:
    """点云处理器"""
    
    def __init__(self, config: Optional[PreprocessingConfig] = None):
        self.config = config or PreprocessingConfig()
    
    def depth_to_pointcloud(self, 
                           frame: RGBDFrame,
                           mask: Optional[np.ndarray] = None) -> PointCloud:
        """
        从深度图生成点云
        
        Args:
            frame: RGB-D图像帧
            mask: 可选掩码，仅处理掩码区域
        """
        depth = frame.depth.copy()
        rgb = frame.rgb
        
        h, w = depth.shape
        
        # 生成像素坐标网格
        u_coords, v_coords = np.meshgrid(np.arange(w), np.arange(h))
        
        # 应用掩码
        if mask is not None:
            valid = mask > 0
        else:
            valid = depth > 0
        
        # 过滤有效点
        u_valid = u_coords[valid]
        v_valid = v_coords[valid]
        z_valid = depth[valid]
        
        # 反投影到相机坐标系
        fx = frame.intrinsic[0, 0]
        fy = frame.intrinsic[1, 1]
        cx = frame.intrinsic[0, 2]
        cy = frame.intrinsic[1, 2]
        
        x_cam = (u_valid - cx) * z_valid / fx
        y_cam = (v_valid - cy) * z_valid / fy
        
        # 转换到机器人基座坐标系
        points_cam = np.stack([x_cam, y_cam, z_valid], axis=1)
        points_world = self._transform_points(points_cam, frame.extrinsic)
        
        # 颜色
        if rgb is not None:
            r = rgb[valid, 0]
            g = rgb[valid, 1]
            b = rgb[valid, 2]
            colors = np.stack([r, g, b], axis=1).astype(np.uint8)
        else:
            colors = None
        
        return PointCloud(
            points=points_world.astype(np.float32),
            colors=colors,
            normals=None,
            timestamp=frame.timestamp
        )
    
    def _transform_points(self, points: np.ndarray, 
                         transform: np.ndarray) -> np.ndarray:
        """应用刚体变换"""
        n = len(points)
        points_h = np.concatenate([points, np.ones((n, 1))], axis=1)
        points_transformed = (transform @ points_h.T).T
        return points_transformed[:, :3]
    
    def downsample_voxel(self, cloud: PointCloud) -> PointCloud:
        """
        体素降采样
        
        使用体素网格过滤器减少点云密度
        """
        if len(cloud.points) == 0:
            return cloud
        
        # 简化的体素降采样
        leaf_size = self.config.voxel_leaf_size
        
        # 计算体素索引
        voxel_indices = np.floor(cloud.points / leaf_size).astype(np.int32)
        
        # 按体素分组
        unique_voxels, inverse_indices = np.unique(
            voxel_indices, axis=0, return_inverse=True
        )
        
        # 计算每个体素的平均值
        n_voxels = len(unique_voxels)
        downsampled_points = np.zeros((n_voxels, 3), dtype=np.float32)
        downsampled_colors = np.zeros((n_voxels, 3), dtype=np.uint8) if cloud.colors is not None else None
        
        for i in range(n_voxels):
            mask = inverse_indices == i
            downsampled_points[i] = cloud.points[mask].mean(axis=0)
            if downsampled_colors is not None and cloud.colors is not None:
                downsampled_colors[i] = cloud.colors[mask].mean(axis=0).astype(np.uint8)
        
        return PointCloud(
            points=downsampled_points,
            colors=downsampled_colors,
            normals=None,
            timestamp=cloud.timestamp
        )
    
    def crop_workspace(self, cloud: PointCloud,
                      min_bound: np.ndarray,
                      max_bound: np.ndarray) -> PointCloud:
        """裁剪到工作空间"""
        mask = np.all(cloud.points >= min_bound, axis=1) & \
               np.all(cloud.points <= max_bound, axis=1)
        
        return PointCloud(
            points=cloud.points[mask],
            colors=cloud.colors[mask] if cloud.colors is not None else None,
            normals=cloud.normals[mask] if cloud.normals is not None else None,
            timestamp=cloud.timestamp
        )
    
    def remove_outliers(self, cloud: PointCloud,
                       k: int = 20,
                       std_threshold: float = 2.0) -> PointCloud:
        """
        统计离群点移除
        
        Args:
            k: 搜索邻点数
            std_threshold: 标准差阈值
        """
        if len(cloud.points) < k + 1:
            return cloud
        
        # 构建KD树
        from scipy.spatial import KDTree
        tree = KDTree(cloud.points)
        
        # 查询k个最近邻
        distances, _ = tree.query(cloud.points, k=k+1)
        mean_distances = distances[:, 1:].mean(axis=1)  # 排除自身
        
        # 计算均值和标准差
        mu = mean_distances.mean()
        sigma = mean_distances.std()
        
        # 移除离群点
        threshold = mu + std_threshold * sigma
        mask = mean_distances < threshold
        
        return PointCloud(
            points=cloud.points[mask],
            colors=cloud.colors[mask] if cloud.colors is not None else None,
            normals=cloud.normals[mask] if cloud.normals is not None else None,
            timestamp=cloud.timestamp
        )


class PerceptionPreprocessor:
    """感知预处理器（整合上述模块）"""
    
    def __init__(self, 
                 preprocessing_config: Optional[PreprocessingConfig] = None,
                 workspace_bounds: Optional[Tuple[np.ndarray, np.ndarray]] = None):
        self.rgbd_processor = RGBDPreprocessor(preprocessing_config)
        self.pcd_processor = PointCloudProcessor(preprocessing_config)
        self.workspace_bounds = workspace_bounds
    
    def process_frame(self, frame: RGBDFrame,
                     focus_roi: Optional[Tuple[int, int, int, int]] = None) -> Tuple[RGBDFrame, PointCloud]:
        """
        完整预处理流程
        
        Returns:
            (processed_frame, pointcloud)
        """
        # 1. 处理深度图
        processed_depth = self.rgbd_processor.process_depth(frame.depth)
        
        # 2. 创建处理后的帧
        processed_frame = RGBDFrame(
            rgb=frame.rgb.copy(),
            depth=processed_depth,
            intrinsic=frame.intrinsic.copy(),
            extrinsic=frame.extrinsic.copy(),
            timestamp=frame.timestamp
        )
        
        # 3. 提取ROI（如果指定）
        mask = None
        if focus_roi is not None:
            mask = np.zeros_like(processed_depth, dtype=np.uint8)
            x_min, y_min, x_max, y_max = focus_roi
            mask[y_min:y_max, x_min:x_max] = 1
        
        # 4. 生成点云
        pointcloud = self.pcd_processor.depth_to_pointcloud(processed_frame, mask)
        
        # 5. 降采样
        pointcloud = self.pcd_processor.downsample_voxel(pointcloud)
        
        # 6. 移除离群点
        pointcloud = self.pcd_processor.remove_outliers(pointcloud)
        
        # 7. 裁剪到工作空间
        if self.workspace_bounds is not None:
            pointcloud = self.pcd_processor.crop_workspace(
                pointcloud, 
                self.workspace_bounds[0], 
                self.workspace_bounds[1]
            )
        
        return processed_frame, pointcloud
```

"""
混合检测引擎
整合YOLOv8闭集检测与YOLO-World开集检测，通过置信度动态切换
"""

from __future__ import annotations
from dataclasses import dataclass, field
from typing import List, Tuple, Optional, Dict, Any
from enum import Enum
import numpy as np
import time

import torch
from ultralytics import YOLO
from dataclasses import dataclass


@dataclass
class BoundingBox:
    """检测框"""
    x1: float
    y1: float
    x2: float
    y2: float
    confidence: float
    class_id: int
    class_name: str
    
    @property
    def center(self) -> Tuple[float, float]:
        return (self.x1 + self.x2) / 2, (self.y1 + self.y2) / 2
    
    @property
    def width(self) -> float:
        return self.x2 - self.x1
    
    @property
    def height(self) -> float:
        return self.y2 - self.y1
    
    @property
    def area(self) -> float:
        return self.width * self.height
    
    def iou_with(self, other: 'BoundingBox') -> float:
        """计算IoU"""
        inter_x1 = max(self.x1, other.x1)
        inter_y1 = max(self.y1, other.y1)
        inter_x2 = min(self.x2, other.x2)
        inter_y2 = min(self.y2, other.y2)
        
        inter_area = max(0, inter_x2 - inter_x1) * max(0, inter_y2 - inter_y1)
        union_area = self.area + other.area - inter_area
        
        return inter_area / union_area if union_area > 0 else 0


@dataclass
class DetectedObject:
    """检测到的目标"""
    bbox: BoundingBox
    source: str                    # "closed_set" / "open_set"
    features: Optional[np.ndarray] = None  # 提取的特征向量


class DetectionMode(Enum):
    """检测模式"""
    CLOSED_SET_ONLY = "closed"      # 仅闭集

class HybridDetectionConfig:
    """混合检测配置"""

    fusion_conf_thresh: float = 0.6   # 融合后置信度阈值

    # 设备

class YOLOv8ClosedSetDetector:
    """
    YOLOv8 闭集检测器
    
    特点：
    - 基于预定义类别库（SKU数据库）
    - 高精度（适合已知品类）
    - 适合标准化货物
    """
    
    def __init__(self, model_path: str, classes: List[str], 
                 conf_thresh: float, device: str):
        self.model_path = model_path
        self.classes = classes
        self.conf_thresh = conf_thresh
        self.device = device
        
        # 加载模型
        self.model = YOLO(model_path)
        self.model.to(device)
    
    def detect(self, image: np.ndarray) -> List[DetectedObject]:
        """
        执行检测
        
        Args:
            image: H x W x 3 RGB图像
            
        Returns:
            检测到的目标列表
        """
        start_time = time.perf_counter()
        
        # 执行推理
        results = self.model.predict(
            image,
            conf=self.conf_thresh,
            classes=list(range(len(self.classes))),  # 所有类别
            verbose=False
        )[0]
        
        detected_objects = []
        
        if results.boxes is not None:
            boxes = results.boxes.xyxy.cpu().numpy()
            confidences = results.boxes.conf.cpu().numpy()
            class_ids = results.boxes.cls.cpu().numpy().astype(int)
            
            for box, conf, cls_id in zip(boxes, confidences, class_ids):
                if cls_id < len(self.classes):
                    bbox = BoundingBox(
                        x1=box[0], y1=box[1],
                        x2=box[2], y2=box[3],
                        confidence=float(conf),
                        class_id=int(cls_id),
                        class_name=self.classes[int(cls_id)]
                    )
                    
                    detected_objects.append(DetectedObject(
                        bbox=bbox,
                        source="closed_set"
                    ))
        
        inference_time = time.perf_counter() - start_time
        
        return detected_objects


class YOLOWorldOpenSetDetector:
    """
    YOLO-World 开集检测器
    
    特点：

    """
    
    def __init__(self, model_path: str,

        self.conf_thresh = conf_thresh

            "cardboard box", "plastic container", "wooden crate",

    def set_prompts(self, prompts: List[str]):

        """
        执行开集检测

            prompts: 检测提示词列表
            
        Returns:
            检测到的目标列表

        # 使用默认提示或自定义提示
        detection_prompts = prompts or self.common_descriptions
        
        # 构建提示文本
        prompt_text = " ".join(detection_prompts)

class DetectionFusionModule:
    """
    检测融合模块
    
    策略：
    1. 闭集检测结果直接采用（高置信度）
    2. 开集检测结果作为补充（填补空白）
    3. 置信度动态加权融合
    4. NMS去重
    """
    
    def __init__(self, config: HybridDetectionConfig):
        self.config = config
    
    def fuse(self, 
             closed_set_results: List[DetectedObject],
             open_set_results: List[DetectedObject]) -> List[DetectedObject]:
        """
        融合两路检测结果
        
        策略：
        - 高置信度闭集结果直接保留
        - 开集结果填补闭集未覆盖区域
        - 使用IoU去重
        """
        all_detections = []
        
        # 处理闭集结果（直接保留）
        for det in closed_set_results:
            # 加权置信度
            weighted_conf = det.bbox.confidence * self.config.closed_set_weight
            fused_det = DetectedObject(
                bbox=BoundingBox(
                    x1=det.bbox.x1, y1=det.bbox.y1,
                    x2=det.bbox.x2, y2=det.bbox.y2,
                    confidence=min(weighted_conf, 1.0),
                    class_id=det.bbox.class_id,
                    class_name=det.bbox.class_name
                ),
                source=det.source,
                features=det.features
            )
            all_detections.append(fused_det)
        
        # 处理开集结果（检查是否需要补充）
        for det in open_set_results:
            # 检查是否与已有闭集结果高度重叠
            should_add = True
            
            for closed_det in closed_set_results:
                iou = det.bbox.iou_with(closed_det.bbox)
                if iou > 0.5:  # 高度重叠，跳过
                    should_add = False
                    break
            
            if should_add:
                # 加权置信度
                weighted_conf = det.bbox.confidence * self.config.open_set_weight
                fused_det = DetectedObject(
                    bbox=BoundingBox(
                        x1=det.bbox.x1, y1=det.bbox.y1,
                        x2=det.bbox.x2, y2=det.bbox.y2,
                        confidence=min(weighted_conf, 1.0),
                        class_id=-1,
                        class_name="unknown"
                    ),
                    source=det.source,
                    features=det.features
                )
                all_detections.append(fused_det)
        
        # NMS去重
        fused_detections = self._nms(all_detections)
        
        # 按置信度排序
        fused_detections.sort(key=lambda x: x.bbox.confidence, reverse=True)
        
        return fused_detections
    
    def _nms(self, detections: List[DetectedObject]) -> List[DetectedObject]:
        """
        非极大值抑制
        
        Args:
            detections: 检测结果列表
            
        Returns:
            去重后的检测结果
        """
        if not detections:
            return []
        
        # 按置信度排序
        sorted_dets = sorted(detections, key=lambda x: x.bbox.confidence, reverse=True)
        
        keep = []
        
        while sorted_dets:
            # 保留最高置信度
            current = sorted_dets.pop(0)
            keep.append(current)
            
            # 移除高度重叠的框
            sorted_dets = [
                det for det in sorted_dets
                if det.bbox.iou_with(current.bbox) < self.config.nms_iou_thresh
            ]
        
        return keep


class HybridDetectionEngine:
    """
    混合检测引擎
    
    整合YOLOv8闭集与YOLO-World开集检测
    """
    
    def __init__(self, config: Optional[HybridDetectionConfig] = None):
        self.config = config or HybridDetectionConfig()
        
        # 初始化检测器
        self.closed_set_detector = YOLOv8ClosedSetDetector(
            model_path=self.config.closed_set_model_path,
            classes=self.config.closed_set_classes,
            conf_thresh=self.config.closed_set_conf_thresh,
            device=self.config.device
        )
        
        self.open_set_detector = YOLOWorldOpenSetDetector(
            model_path=self.config.open_set_model_path,
            conf_thresh=self.config.open_set_conf_thresh,
            device=self.config.device
        )
        
        # 初始化融合模块
        self.fusion_module = DetectionFusionModule(self.config)
        
        # 性能统计
        self._stats = {
            'total_inferences': 0,
            'closed_set_time': [],
            'open_set_time': [],
            'fusion_time': []
        }
    
    def detect(self, 
               image: np.ndarray,
               mode: DetectionMode = DetectionMode.HYBRID,
               open_set_prompts: Optional[List[str]] = None) -> List[DetectedObject]:
        """
        执行混合检测
        
        Args:
            image: H x W x 3 RGB图像
            mode: 检测模式
            open_set_prompts: 开集检测提示词
            
        Returns:
            融合后的检测结果
        """
        self._stats['total_inferences'] += 1
        
        closed_results = []
        open_results = []
        
        if mode in [DetectionMode.CLOSED_SET_ONLY, DetectionMode.HYBRID]:
            # 闭集检测
            start = time.perf_counter()
            closed_results = self.closed_set_detector.detect(image)
            self._stats['closed_set_time'].append(time.perf_counter() - start)
        
        if mode in [DetectionMode.OPEN_SET_ONLY, DetectionMode.HYBRID]:
            # 开集检测
            start = time.perf_counter()
            self.open_set_detector.set_prompts(open_set_prompts or [])
            open_results = self.open_set_detector.detect(image)
            self._stats['open_set_time'].append(time.perf_counter() - start)
        
        # 融合
        start = time.perf_counter()
        if mode == DetectionMode.HYBRID:
            fused_results = self.fusion_module.fuse(closed_results, open_results)
        elif mode == DetectionMode.CLOSED_SET_ONLY:
            fused_results = closed_results
        else:
            fused_results = open_results
        
        # 过滤低置信度
        fused_results = [
            det for det in fused_results
            if det.bbox.confidence >= self.config.fusion_conf_thresh
        ]
        
        self._stats['fusion_time'].append(time.perf_counter() - start)
        
        return fused_results
    
    def get_stats(self) -> Dict[str, Any]:
        """获取性能统计"""
        import numpy as np
        
        def safe_mean(arr):
            return np.mean(arr) if arr else 0.0
        
        return {
            'total_inferences': self._stats['total_inferences'],
            'avg_closed_set_time_ms': safe_mean(self._stats['closed_set_time']) * 1000,
            'avg_open_set_time_ms': safe_mean(self._stats['open_set_time']) * 1000,
            'avg_fusion_time_ms': safe_mean(self._stats['fusion_time']) * 1000,
            'avg_total_time_ms': (
                safe_mean(self._stats['closed_set_time']) +
                safe_mean(self._stats['open_set_time']) +
                safe_mean(self._stats['fusion_time'])
            ) * 1000
        }
    
    def reset_stats(self):
        """重置统计"""
        self._stats = {
            'total_inferences': 0,
            'closed_set_time': [],
            'open_set_time': [],
            'fusion_time': []
        }

@dataclass
class Object6DPose:
    """物体6D位姿"""
    object_id: str
    class_name: str
    rotation: np.ndarray              # 3x3 旋转矩阵
    translation: np.ndarray           # 3维平移向量
    confidence: float
    
    @property
    def position(self) -> np.ndarray:
        return self.translation.copy()
    
    def to_pose6d(self) -> Pose6D:
        """转换为Pose6D格式"""
        from scipy.spatial.transform import Rotation
        return Pose6D(
            position=self.translation,
            orientation=Rotation.from_matrix(self.rotation).as_quat()
        )
    
    def to_matrix(self) -> np.ndarray:
        """转换为4x4变换矩阵"""

        matrix[:3, :3] = self.rotation
        matrix[:3, 3] = self.translation
        return matrix

class SimpleLocalGraspPlanner:
    """
    本地抓取规划器
    
    提供轻量级本地抓取规划（云端规划的降级方案）
    """
    
    def __init__(self, gripper_max_width: float = 0.1):
        self.gripper_max_width = gripper_max_width
    
    def plan_grasp(self,
                  pointcloud: PointCloud,
                  object_center: np.ndarray,
                  object_bounds: Tuple[float, float, float]) -> List[GraspPose]:
        """
        本地抓取规划
        
        简化策略：
        1. 从点云估计物体尺寸
        2. 选择顶面中心作为抓取点
        3. 垂直接近
        """
        grasps = []
        
        # 找顶面点
        z_max = pointcloud.points[:, 2].max()
        top_mask = pointcloud.points[:, 2] > z_max - 0.05
        top_points = pointcloud.points[top_mask]
        
        if len(top_points) == 0:
            return grasps
        
        # 计算抓取位置（顶面中心）
        grasp_center = top_points.mean(axis=0)
        
        # 估计物体尺寸
        obj_extent = pointcloud.points.max(axis=0) - pointcloud.points.min(axis=0)
        max_dim = max(obj_extent)
        gripper_width = min(max_dim * 1.2, self.gripper_max_width)
        
        grasp_point = GraspPoint(
            position=grasp_center,
            approach_direction=np.array([0, 0, -1]),  # 从上方接近
            gripper_width=gripper_width,
            score=0.7,  # 本地规划置信度较低
            rotation=None
        )
        
        # 构建抓取姿态
        from scipy.spatial.transform import Rotation
        
        # 预抓取姿态（上方）
        pre_pos = grasp_center + np.array([0, 0, 0.15])
        pre_grasp = Pose6D(
            position=pre_pos,
            orientation=Rotation.from_euler('xyz', [0, 0, 0]).as_quat()
        )
        
        # 抓取姿态
        grasp = Pose6D(
            position=grasp_center,
            orientation=Rotation.from_euler('xyz', [0, 0, 0]).as_quat()
        )
        
        # 抓取后姿态
        post_pos = grasp_center + np.array([0, 0, 0.3])
        post_grasp = Pose6D(
            position=post_pos,
            orientation=Rotation.from_euler('xyz', [0, 0, 0]).as_quat()
        )
        
        grasps.append(GraspPose(
            object_id="local_estimated",
            grasp_point=grasp_point,
            pre_grasp_pose=pre_grasp,
            grasp_pose=grasp,
            post_grasp_pose=post_grasp,
            score=0.7,
            method='local'
        ))
        
        return grasps


class PerceptionCoordinator:
    """
    感知协调器
    
    管理本地与云端感知的协调，根据网络状态和置信度动态切换
    """
    
    def __init__(self,
                 cloud_endpoint: str = "http://cloud-robotics.local:8080",
                 use_cloud_fallback: bool = True):
        self.cloud_client = CloudServiceClient(endpoint=cloud_endpoint)
        self.local_grasp_planner = SimpleLocalGraspPlanner()
        self.use_cloud_fallback = use_cloud_fallback
        
        # 网络状态
        self._cloud_available = True
        self._last_health_check = 0.0
    
    def check_cloud_health(self) -> bool:
        """检查云端服务健康状态"""
        current_time = time.time()
        
        # 缓存结果，1分钟内不重复检查
        if current_time - self._last_health_check < 60:
            return self._cloud_available
        
        try:
            response = requests.get(
                f"{self.cloud_client.endpoint}/health",
                timeout=2.0
            )
            self._cloud_available = response.status_code == 200
        except:
            self._cloud_available = False
        
        self._last_health_check = current_time
        return self._cloud_available
    
    def estimate_and_grasp(self,
                          frame: RGBDFrame,
                          pointcloud: PointCloud,
                          detections: List,  # HybridDetection输出
                          min_confidence: float = 0.8) -> List[GraspPose]:
        """
        协调感知流程
        
        策略：
        1. 高置信度检测 -> 直接本地规划
        2. 低置信度/复杂场景 -> 云端处理
        3. 网络异常 -> 本地降级
        """
        all_grasps = []
        
        for detection in detections:
            bbox = detection.bbox
            
            # 检查置信度
            if detection.bbox.confidence >= min_confidence and detection.source == "closed_set":
                # 高置信度闭集检测 -> 本地快速规划
                obj_center = np.array([
                    (bbox.x1 + bbox.x2) / 2,
                    (bbox.y1 + bbox.y2) / 2,
                    0
                ])
                
                # 映射到3D
                depth_val = frame.depth[
                    int(obj_center[1]), int(obj_center[0])
                ] / 1000.0  # 转米
                
                if depth_val > 0:
                    fx, fy = frame.intrinsic[0, 0], frame.intrinsic[1, 1]
                    cx, cy = frame.intrinsic[0, 2], frame.intrinsic[1, 2]
                    
                    obj_center_3d = np.array([
                        (obj_center[0] - cx) * depth_val / fx,
                        (obj_center[1] - cy) * depth_val / fy,
                        depth_val
                    ])
                    
                    # 本地抓取规划
                    grasps = self.local_grasp_planner.plan_grasp(
                        pointcloud=pointcloud,
                        object_center=obj_center_3d,
                        object_bounds=(bbox.width, bbox.height, 0.1)
                    )
                    all_grasps.extend(grasps)
            else:
                # 低置信度或开集检测 -> 云端处理
                if self.check_cloud_health() and self.use_cloud_fallback:
                    cloud_grasps = self._cloud_grasp_planning(
                        frame, pointcloud, detection
                    )
                    all_grasps.extend(cloud_grasps)
                else:
                    # 本地降级
                    grasps = self.local_grasp_planner.plan_grasp(
                        pointcloud=pointcloud,
                        object_center=np.array([0, 0, 0]),
                        object_bounds=(0.1, 0.1, 0.1)
                    )
                    all_grasps.extend(grasps)
        
        # 按分数排序
        all_grasps.sort(key=lambda x: x.score, reverse=True)
        
        return all_grasps
    
    def _cloud_grasp_planning(self,
                             frame: RGBDFrame,
                             pointcloud: PointCloud,
                             detection) -> List[GraspPose]:
        """调用云端抓取规划"""
        # 首先请求云端姿态估计
        object_class = detection.bbox.class_name
        
        try:
            poses = self.cloud_client.estimate_pose(
                image=frame.rgb,
                pointcloud=pointcloud.points,
                object_class=object_class
            )
            
            if poses:
                # 请求抓取规划
                return self.cloud_client.plan_grasp(
                    pointcloud=pointcloud.points,
                    object_poses=poses
                )
        except Exception as e:
            print(f"Cloud processing failed: {e}")
        
        return []

```python
# src/decision/task_scheduler.py
"""
任务调度器
负责任务分解、序列优化、执行监控
"""

from __future__ import annotations
from dataclasses import dataclass, field
from typing import List, Tuple, Optional, Dict, Callable
from enum import Enum
import numpy as np
import heapq
import time


class TaskPriority(Enum):
    """任务优先级"""
    CRITICAL = 1    # 紧急任务
    HIGH = 2        # 高优先级
    NORMAL = 3      # 普通
    LOW = 4         # 低优先级


@dataclass
class Task:
    """任务"""
    task_id: str
    task_type: str                        # "pick" / "place" / "move"
    source_pose: Optional[np.ndarray]      # 源位置
    target_pose: Optional[np.ndarray]     # 目标位置
    object_class: Optional[str] = None    # 目标物体类别
    priority: TaskPriority = TaskPriority.NORMAL
    dependencies: List[str] = field(default_factory=list)  # 前置任务ID
    estimated_duration: float = 5.0       # 估计执行时间（秒）
    created_time: float = field(default_factory=time.time)
    
    def __lt__(self, other):
        """支持优先队列排序"""
        return self.priority.value < other.priority.value


@dataclass
class ExecutionStatus:
    """执行状态"""
    task_id: str
    status: str                           # "pending" / "running" / "completed" / "failed"
    start_time: Optional[float] = None
    end_time: Optional[float] = None
    progress: float = 0.0                  # 0-1
    error_message: Optional[str] = None


class TaskScheduler:
    """
    任务调度器
    
    功能：
    1. 任务队列管理
    2. 任务依赖解析
    3. 批次优化
    4. 执行监控与重试
    """
    
    def __init__(self, max_concurrent: int = 1):
        self.max_concurrent = max_concurrent
        self.task_queue: List[Task] = []
        self.executing_tasks: Dict[str, Task] = {}
        self.completed_tasks: Dict[str, ExecutionStatus] = {}
        self.failed_tasks: Dict[str, ExecutionStatus] = {}
        
        # 执行回调
        self._execute_callback: Optional[Callable] = None
        
        # 统计
        self._stats = {
            'total_tasks': 0,
            'completed_tasks': 0,
            'failed_tasks': 0,
            'avg_execution_time': 0.0
        }
    
    def add_task(self, task: Task):
        """添加任务"""
        self.task_queue.append(task)
        heapq.heapify(self.task_queue)
        self._stats['total_tasks'] += 1
    
    def add_tasks(self, tasks: List[Task]):
        """批量添加任务"""
        for task in tasks:
            self.add_task(task)
    
    def get_ready_tasks(self) -> List[Task]:
        """获取可执行的任务（依赖已满足）"""
        ready = []
        for task in self.task_queue:
            # 检查依赖
            deps_satisfied = all(
                tid in self.completed_tasks 
                for tid in task.dependencies
            )
            if deps_satisfied:
                ready.append(task)
        return ready
    
    def get_next_batch(self, batch_size: Optional[int] = None) -> List[Task]:
        """
        获取下一批次任务
        
        Args:
            batch_size: 批次大小，None表示max_concurrent
            
        Returns:
            可执行的任务列表
        """
        if batch_size is None:
            batch_size = self.max_concurrent
        
        # 检查当前执行中的任务数
        available_slots = self.max_concurrent - len(self.executing_tasks)
        if available_slots <= 0:
            return []
        
        # 获取就绪任务
        ready_tasks = self.get_ready_tasks()
        
        # 按优先级排序并返回
        ready_tasks.sort(key=lambda t: (t.priority.value, t.created_time))
        batch = ready_tasks[:batch_size]
        
        # 从队列移除
        for task in batch:
            self.task_queue.remove(task)
        
        return batch
    
    def start_execution(self, tasks: List[Task], 
                       execute_callback: Callable[[Task], bool]):
        """
        开始执行任务
        
        Args:
            tasks: 要执行的任务
            execute_callback: 执行回调，返回是否成功
        """
        self._execute_callback = execute_callback
        
        for task in tasks:
            self.executing_tasks[task.task_id] = task
            self.completed_tasks[task.task_id] = ExecutionStatus(
                task_id=task.task_id,
                status="running",
                start_time=time.time()
            )
    
    def complete_task(self, task_id: str, success: bool, 
                     error_message: Optional[str] = None):
        """完成任务"""
        if task_id not in self.executing_tasks:
            return
        
        task = self.executing_tasks.pop(task_id)
        
        status = self.completed_tasks[task_id]
        status.status = "completed" if success else "failed"
        status.end_time = time.time()
        status.progress = 1.0 if success else status.progress
        status.error_message = error_message
        
        if success:
            self._stats['completed_tasks'] += 1
        else:
            self.failed_tasks[task_id] = status
            self._stats['failed_tasks'] += 1
    
    def update_progress(self, task_id: str, progress: float):
        """更新任务进度"""
        if task_id in self.completed_tasks:
            self.completed_tasks[task_id].progress = progress
    
    def get_execution_stats(self) -> Dict:
        """获取执行统计"""
        total_completed = self._stats['completed_tasks'] + self._stats['failed_tasks']
        
        return {
            'total_tasks': self._stats['total_tasks'],
            'pending_tasks': len(self.task_queue),
            'executing_tasks': len(self.executing_tasks),
            'completed_tasks': self._stats['completed_tasks'],
            'failed_tasks': self._stats['failed_tasks'],
            'completion_rate': self._stats['completed_tasks'] / total_completed if total_completed > 0 else 0
        }
    
    def optimize_sequence(self, tasks: List[Task]) -> List[Task]:
        """
        优化任务执行序列
        
        策略：
        1. 按依赖关系拓扑排序
        2. 同层任务按优先级+位置优化
        3. 空间局部性：优先执行位置接近的任务
        """
        # 构建依赖图
        task_map = {t.task_id: t for t in tasks}
        in_degree = {t.task_id: len(t.dependencies) for t in tasks}
        adj_list = {t.task_id: [] for t in tasks}
        
        for t in tasks:
            for dep in t.dependencies:
                if dep in task_map:
                    adj_list[dep].append(t.task_id)
        
        # Kahn算法拓扑排序
        queue = [t.task_id for t in tasks if in_degree[t.task_id] == 0]
        sorted_ids = []
        
        while queue:
            # 按优先级排序
            queue.sort(key=lambda tid: (
                task_map[tid].priority.value,
                task_map[tid].created_time
            ))
            
            current = queue.pop(0)
            sorted_ids.append(current)
            
            for neighbor in adj_list[current]:
                in_degree[neighbor] -= 1
                if in_degree[neighbor] == 0:
                    queue.append(neighbor)
        
        # 检查循环依赖
        if len(sorted_ids) != len(tasks):
            print("Warning: Circular dependency detected!")
            return tasks
        
        return [task_map[tid] for tid in sorted_ids]


class BatchOptimizer:
    """
    批次优化器
    
    优化多任务执行顺序，减少总执行时间
    """
    
    def __init__(self, distance_weight: float = 0.3, 
                 priority_weight: float = 0.7):
        self.distance_weight = distance_weight
        self.priority_weight = priority_weight
    
    def optimize(self, tasks: List[Task]) -> List[Task]:
        """
        优化批次顺序
        
        目标函数：
        min (α * 移动距离 + β / 优先级)
        
        使用贪心+局部搜索优化
        """
        if len(tasks) <= 1:
            return tasks
        
        # 简化的贪心优化
        optimized = []
        remaining = tasks.copy()
        current_pose = np.zeros(3)
        
        while remaining:
            # 计算每个任务的得分
            scored = []
            
            for task in remaining:
                # 位置得分（越近越好）
                if task.source_pose is not None:
                    distance = np.linalg.norm(task.source_pose - current_pose)
                else:
                    distance = 0
                
                # 优先级得分
                priority_score = 1.0 / task.priority.value
                
                # 综合得分
                score = (self.distance_weight * distance + 
                        self.priority_weight * (1 - priority_score))
                
                scored.append((score, task))
            
            # 选择得分最低的
            scored.sort(key=lambda x: x[0])
            chosen = scored[0][1]
            optimized.append(chosen)
            remaining.remove(chosen)
            
            # 更新当前位置
            if chosen.target_pose is not None:
                current_pose = chosen.target_pose
            elif chosen.source_pose is not None:
                current_pose = chosen.source_pose
        
        return optimized


class ExecutionMonitor:
    """
    执行监控器
    
    监控任务执行状态，处理异常和重试
    """
    
    def __init__(self, max_retries: int = 3):
        self.max_retries = max_retries
        self.retry_count: Dict[str, int] = {}
        self._monitoring = False
    
    def should_retry(self, task_id: str, error: str) -> bool:
        """检查是否应该重试"""
        retries = self.retry_count.get(task_id, 0)
        
        # 只对可恢复错误重试
        recoverable_errors = [
            "timeout", "network", "collision_temporary", 
            "sensor_noisy", "object_moved"
        ]
        
        is_recoverable = any(e in error.lower() for e in recoverable_errors)
        
        return retries < self.max_retries and is_recoverable
    
    def record_retry(self, task_id: str):
        """记录重试"""
        self.retry_count[task_id] = self.retry_count.get(task_id, 0) + 1
    
    def get_retry_count(self, task_id: str) -> int:
        """获取重试次数"""
        return self.retry_count.get(task_id, 0)
    
    def check_health(self, status: ExecutionStatus) -> Tuple[bool, Optional[str]]:
        """
        健康检查
        
        Returns:
            (is_healthy, error_message)
        """
        # 超时检查
        if status.status == "running" and status.start_time is not None:
            elapsed = time.time() - status.start_time
            if elapsed > 60:  # 60秒超时
                return False, "Execution timeout"
        
        # 进度检查
        if status.status == "running" and status.progress == 0:
            elapsed = time.time() - status.start_time if status.start_time else 0
            if elapsed > 30:
                return False, "No progress detected"
        
        return True, None
```
